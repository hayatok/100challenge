const COLLISION_CHUNK: float = 64.0
# Authored islands: each horizontal transfer asks the player to turn in, then
# unwind before the next jump. Landing platforms are straight and visible.
const KNOTS: Array[Vector2] = [Vector2(0, 0), Vector2(35, 0), Vector2(135, 34),
	Vector2(175, 34), Vector2(225, 18), Vector2(280, 18), Vector2(390, -26),
	Vector2(430, -26), Vector2(495, 8), Vector2(565, 8), Vector2(615, 8),
	Vector2(705, -32), Vector2(785, 12), Vector2(930, 12)]
const GAPS: Array[Vector2] = [Vector2(245, 257), Vector2(530, 546), Vector2(825, 849)]
const LAUNCH_LENGTH: float = 18.0
const LAUNCH_HEIGHT: float = 1.5
# Marks start inside the measured Space window at each intended approach speed.
const LAUNCH_MARK_LENGTHS: Array[float] = [10.0, 8.0, 14.0]
const CHECKPOINTS: Array[float] = [0.0, 165.0, 280.0, 430.0, 565.0, 615.0, 785.0]


func _ready() -> void:
	process_mode = Node.PROCESS_MODE_PAUSABLE
	_build_course()


func center_at(s: float) -> Vector3:
	var x: float = KNOTS.back().y
	if s <= KNOTS.front().x:
		x = KNOTS.front().y
	else:
		for index in range(KNOTS.size() - 1):
			var a := KNOTS[index]
			var b := KNOTS[index + 1]
			if s <= b.x:
				x = lerpf(a.y, b.y, smoothstep(a.x, b.x, s))
				break
	# Downhill terraces create an actual change in height without hiding landings.
	var y := 24.0 - s * 0.014 - 3.0 * smoothstep(280.0, 430.0, s)
	y -= 3.0 * smoothstep(615.0, 785.0, s)
	y += launch_height_at(s)
	return Vector3(x, y, -s)


func launch_height_at(s: float) -> float:
	for gap in GAPS:
		if s >= gap.x - LAUNCH_LENGTH and s < gap.x:
			return LAUNCH_HEIGHT * smoothstep(gap.x - LAUNCH_LENGTH, gap.x, s)
		if s >= gap.x and s < gap.y:
			# Camera/reference height continues over the void without a lip jump.
			return LAUNCH_HEIGHT * (1.0 - smoothstep(gap.x, gap.y, s))
	return 0.0


func tangent_at(s: float) -> Vector3:
	return (center_at(s + 0.25) - center_at(s - 0.25)).normalized()


func heading_at(s: float) -> float:
	var tangent := tangent_at(s)
	return atan2(-tangent.x, -tangent.z)


func curvature_at(s: float) -> float:
	return angle_difference(heading_at(s - 2.0), heading_at(s + 2.0)) / 4.0


func half_width_at(s: float) -> float:
	return lerpf(9.0, base_half_width, smoothstep(120.0, 300.0, s))


func surface_at(s: float, offset: float, lift: float = 0.0) -> Vector3:
	return _surface_point(s, offset, lift)


func checkpoint_at(s: float) -> float:
	var checkpoint := 0.0
	for candidate in CHECKPOINTS:
		if candidate <= s:
			checkpoint = candidate
	return checkpoint


func gap_at(s: float) -> Vector2:
	for gap in GAPS:
		if s >= gap.x and s < gap.y:
			return gap
	return Vector2.ZERO


func next_gap_at(s: float) -> Vector2:

			return gap
	return Vector2.ZERO

func _build_course() -> void:
	var road := _new_surface()
	var guides := _new_surface()
	var launch := _new_surface()
	var collision_faces := PackedVector3Array()
	var chunk_start := -12.0
	var s := chunk_start
	var end := length + 24.0
	while s < end:
		var next := minf(s + STEP, end)
		if gap_at((s + next) * 0.5) == Vector2.ZERO:
			var left_a := -half_width_at(s)
			var right_a := half_width_at(s)
			var left_b := -half_width_at(next)
			var right_b := half_width_at(next)
			_emit_strip(road, collision_faces, s, next, left_a, right_a, left_b, right_b)
			_emit_strip(guides, PackedVector3Array(), s, next,
				left_a, left_a + 0.18, left_b, left_b + 0.18, 0.025)
			_emit_strip(guides, PackedVector3Array(), s, next,
				right_a - 0.18, right_a, right_b - 0.18, right_b, 0.025)
			# Repeated transverse launch marks reveal the takeoff zone and scale.
			for gap_index in GAPS.size():
				var gap := GAPS[gap_index]
				if s >= gap.x - LAUNCH_MARK_LENGTHS[gap_index] and s < gap.x and int(s) % 3 == 0:
					_emit_strip(launch, PackedVector3Array(), s, next, left_a, right_a, left_b, right_b, 0.035)
				if s >= gap.y and s < gap.y + 4.0:
					_emit_strip(launch, PackedVector3Array(), s, next, left_a, right_a, left_b, right_b, 0.035)
		s = next

	_add_visual(guides, "RouteGuides", Color("168b8c"))

func _new_surface() -> SurfaceTool:
	var surface := SurfaceTool.new()
	surface.begin(Mesh.PRIMITIVE_TRIANGLES)
	return surface


func _surface_point(s: float, offset: float, rise: float = 0.0) -> Vector3:
	# Cross sections remain on z = -s so progress, gaps and checkpoints agree.
	var bank := clampf(-heading_at(s) * 0.18, -0.12, 0.12)
	return center_at(s) + Vector3(offset, offset * tan(bank) + rise, 0.0)


func _emit_strip(surface: SurfaceTool, faces: PackedVector3Array,
		s: float, next: float, left_a: float, right_a: float,
		left_b: float, right_b: float, lift: float = 0.0) -> void:
	var rise_a := lift
	var rise_b := lift
	var a := _surface_point(s, left_a, rise_a)
	var b := _surface_point(s, right_a, rise_a)
	var c := _surface_point(next, right_b, rise_b)
	var d := _surface_point(next, left_b, rise_b)
	_emit_triangle(surface, faces, a, c, b)
	_emit_triangle(surface, faces, a, d, c)


func _emit_triangle(surface: SurfaceTool, faces: PackedVector3Array,
		a: Vector3, b: Vector3, c: Vector3) -> void:
	# Godot uses clockwise front faces; this order exposes the road from above.
	var normal := (c - a).cross(b - a).normalized()
	for vertex in [a, b, c]:
		surface.set_normal(normal)
		surface.add_vertex(vertex)
		faces.append(vertex)


func _add_collision(faces: PackedVector3Array, start: float) -> void:
	var shape := ConcavePolygonShape3D.new()
	shape.set_faces(faces)
	shape.backface_collision = false
	var body := StaticBody3D.new()
	body.name = "RoadCollision_%04d" % int(start + 12.0)
	body.collision_layer = 2
	body.collision_mask = 1
	var collider := CollisionShape3D.new()
	collider.shape = shape
	body.add_child(collider)
	add_child(body)


func _add_visual(surface: SurfaceTool, node_name: String, color: Color) -> void:
	var material := StandardMaterial3D.new()
	material.albedo_color = color
	material.roughness = 0.88
	var instance := MeshInstance3D.new()
	instance.name = node_name
	instance.mesh = surface.commit()
	instance.material_override = material
	if node_name == "MainSurface":
		var stone := ShaderMaterial.new()
		stone.shader = preload("res://shaders/stone_path.gdshader")
		stone.set_shader_parameter("stone_color", Color("a4a08d") if node_name == "MainSurface" else Color("918f7d"))
		instance.material_override = stone
	add_child(instance)

		return

	progress = maxf(0.0, -global_position.z)

	# into a free ballistic launch before Space reaches the visible lip.

	var on_platform := true

	if course.has_method("gap_at"):

		if carve_charge > 0.0 and signf(steer_amount) != _carve_direction:
			# Changing edges discards the previous load, so wiggling cannot stack it.
			carve_charge = 0.0
			_carve_time = 0.0
			_carve_distance = 0.0
		_carve_direction = signf(steer_amount)
		_carve_time += delta
		_carve_distance += maxf(0.0, -velocity.z) * delta
		var curvature := 0.0
		if course.has_method("curvature_at"):
			curvature = float(course.call("curvature_at", progress))
		var aligned := curvature * -steer_amount > 0.001
		charge_rate = (aligned_charge_rate if aligned else carve_charge_rate) * absf(steer_amount)
		if absf(lateral) > float(course.call("half_width_at", progress)):
			charge_rate = 0.0
		var gained := minf(1.0 - carve_charge, charge_rate * delta)
		carve_charge += gained
	elif _carve_release_armed and not Input.is_action_pressed("brake") and carve_charge > 0.0:
		var released := carve_charge
		var held_edge := _carve_time >= minimum_carve_time and _carve_distance >= minimum_carve_distance
		carve_charge = 0.0
		_carve_release_armed = false
		_carve_time = 0.0
		_carve_distance = 0.0
		if held_edge and released >= 0.22:
			# Reward the committed release once, never partial taps or reversals.
			grant_energy(released * 12.0)
			carve_released.emit(released)
			_start_burst(released)
	elif carve_charge > 0.0 and not grounded:
		carve_charge = maxf(0.0, carve_charge - 0.15 * delta)


func _start_burst(strength: float) -> void:
	_burst_strength = clampf(strength, 0.0, 1.0)
	burst_left = maxf(burst_left, 0.55 + _burst_strength * 1.1)
	boosting = true
	# The release should be felt now, rather than several seconds later.
	speed = maxf(speed, lerpf(29.0, 36.0, _burst_strength))
	burst_started.emit(_burst_strength)


func _update_speed(delta: float, tangent: Vector3, grounded: bool) -> void:
	var target := cruise_speed
	var acceleration := 7.0
	if grounded:
		target = minf(downhill_speed, cruise_speed + maxf(0.0, -tangent.y) * 35.0)
	if boosting:
		target = boost_speed
		acceleration = 20.0
	if carving:
		target = 20.0
		acceleration = 13.0
	elif grounded and _brake_ready and Input.is_action_pressed("brake"):
		target = brake_speed
		acceleration = 18.0
	speed = move_toward(speed, target, acceleration * delta)


func _update_progress() -> void:
	max_progress = maxf(max_progress, progress)


func _update_landing(delta: float, was_grounded: bool) -> void:
	if not is_on_floor():
		_air_time += delta
		airborne_time = _air_time
		return
	if not was_grounded and not _suppress_landing and _air_time >= 0.10:
		landing_strength = clampf(_air_time / 0.85, 0.0, 1.0)
		landed.emit()
		if absf(angle_difference(heading, _heading_at(progress))) < 0.22 and _air_time >= 0.30:
			grant_energy(14.0)
			clean_landed.emit(landing_strength)
	_air_time = 0.0
	airborne_time = 0.0
	_suppress_landing = false


func _update_visual(delta: float, tangent: Vector3, steer: float) -> void:
	var visual_heading := heading - carve_amount * 0.15
	_visual.rotation.y = lerp_angle(_visual.rotation.y, visual_heading, 1.0 - exp(-8.0 * delta))
	_visual.rotation.z = lerpf(_visual.rotation.z, -steer * (0.42 if carving else 0.24), 1.0 - exp(-9.0 * delta))
	var pitch := atan2(tangent.y, Vector2(tangent.x, tangent.z).length()) if is_on_floor() else 0.0
	_visual.rotation.x = lerpf(_visual.rotation.x, pitch, 1.0 - exp(-7.0 * delta))


func _center_at(s: float) -> Vector3:
	if is_instance_valid(course):
		return course.call("center_at", s)
	return Vector3(0.0, 0.0, -s)


func _tangent_at(s: float) -> Vector3:
	var tangent: Vector3 = course.call("tangent_at", s)
	return tangent.normalized() if tangent.length_squared() > 0.0001 else Vector3.FORWARD


func _heading_at(s: float) -> float:
	if not is_instance_valid(course):
		return 0.0
	var tangent := _tangent_at(s)
	return atan2(-tangent.x, -tangent.z)

func _process(delta: float) -> void:
	if not is_instance_valid(rider) or not is_instance_valid(course):
		return
	if title_view:
		reset_camera()
		return
	if not _initialized or rider.global_position.distance_to(_previous_rider_position) > 20.0:
		reset_camera()
		return
	_previous_rider_position = rider.global_position
	_heading = lerp_angle(_heading, _rider_heading(), 1.0 - exp(-3.8 * delta))
	_boost_impulse = move_toward(_boost_impulse, 0.0, delta * 3.3)
	_land_impulse = move_toward(_land_impulse, 0.0, delta * 5.0)
	_hit_impulse = move_toward(_hit_impulse, 0.0, delta * 4.0)
	var rider_position := rider.get_global_transform_interpolated().origin
	if rider_position.distance_to(_previous_render_position) > 20.0:
		reset_camera()
		return
	var ground: Vector3 = course.call("center_at", maxf(0.0, -rider_position.z))
	# Carry translation at the same interpolated instant as the rendered rider.
	# Smooth only the relative orbit/height; world-space damping adds speed / response lag.
	var jump_height := clampf(rider_position.y - ground.y, 0.0, 7.0)
	var jump_change := jump_height - _previous_jump_height
	var translation := rider_position - _previous_render_position
	translation.y = ground.y - _previous_ground_height
	global_position += translation + Vector3.UP * jump_change * 0.75
	_look_target += translation + Vector3.UP * jump_change * 0.18

	_previous_ground_height = ground.y
	_previous_jump_height = jump_height

	fov = 62.0 if reduced_motion else lerpf(fov, target_fov, 1.0 - exp(-4.5 * delta))


func _framing(rider_position: Vector3) -> Array[Vector3]:
	if title_view:
		var offset := Vector3(3.6, 1.65, 1.2)
		var camera_right := Vector3(-offset.x, 0.0, -offset.z).normalized().cross(Vector3.UP)
		return [rider_position + offset, rider_position + Vector3.UP - camera_right * 1.2]
	var s := maxf(0.0, -rider_position.z)
	var center: Vector3 = course.call("center_at", s)
	var forward := Vector3(-sin(_heading), 0.0, -cos(_heading))
	var right := forward.cross(Vector3.UP).normalized()
	var speed_ratio := clampf((_number("speed") - 18.0) / 20.0, 0.0, 1.0)
	var impulse := 0.0 if reduced_motion else _boost_impulse * 0.08
	var position_target := rider_position - forward * (follow_distance + impulse)
	var jump_height := clampf(rider_position.y - center.y, 0.0, 7.0)
	# Track airborne height enough to preserve the pose; a low gaze keeps the board above HUD.
	position_target.y = center.y + follow_height + jump_height * 0.75
	if not reduced_motion:
		position_target.y -= _land_impulse * 0.10
		position_target += right * _hit_impulse * 0.08

	var player_target := rider_position + forward * look_ahead
	var target := player_target.lerp(path_target, 0.27)
	target.y = lerpf(center.y, path_target.y, 0.72) - 1.1 + jump_height * 0.18
	return [position_target, target]

	var tangent: Vector3 = course.call("tangent_at", maxf(0.0, -rider.global_position.z))
	return atan2(-tangent.x, -tangent.z)

	var value = rider.get(property)
